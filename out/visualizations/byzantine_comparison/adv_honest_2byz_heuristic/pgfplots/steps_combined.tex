\definecolor{colorNoByzantine}{HTML}{E41A1C}
\definecolor{colorUserDirLie}{HTML}{377EB8}

\pgfplotsset{
    % Shared styles
    ci path/.style={draw=none},
    ci fill/.style={opacity=0.15},
    mean/.style={thick, mark repeat=200, mark size=1.5pt},
    % Per-condition styles
    NoByzantine/.style={color=colorNoByzantine, mark=*},
    UserDirLie/.style={color=colorUserDirLie, mark=square*},
}

\begin{tikzpicture}
    \begin{axis}[
        grid=major,
        grid style={dashed, gray!30},
        width=\linewidth,
        xlabel={\small $t$},
        ylabel={\small $\tau$ [\si{\mega\bit\per\second}]},
        legend style={at={(0.5,-0.20)}, anchor=north, legend columns=3},
        tick label style={font=\small},
        label style={font=\small},
    ]

    % ===== No Byzantine =====
    \addplot [name path=NoByzantine upper, ci path, forget plot] table [x=x, y=upNoByzantine, col sep=comma] {steps_combined.csv};
    \addplot [name path=NoByzantine lower, ci path, forget plot] table [x=x, y=lowNoByzantine, col sep=comma] {steps_combined.csv};
    \addplot [NoByzantine, ci fill, forget plot] fill between [of=NoByzantine upper and NoByzantine lower];
    \addplot [NoByzantine, mean] table [x=x, y=meanNoByzantine, col sep=comma] {steps_combined.csv};
    \addlegendentry{No Byzantine}

    % ===== User Dir Lie =====
    \addplot [name path=UserDirLie upper, ci path, forget plot] table [x=x, y=upUserDirLie, col sep=comma] {steps_combined.csv};
    \addplot [name path=UserDirLie lower, ci path, forget plot] table [x=x, y=lowUserDirLie, col sep=comma] {steps_combined.csv};
    \addplot [UserDirLie, ci fill, forget plot] fill between [of=UserDirLie upper and UserDirLie lower];
    \addplot [UserDirLie, mean] table [x=x, y=meanUserDirLie, col sep=comma] {steps_combined.csv};
    \addlegendentry{User Dir Lie}

    \end{axis}
\end{tikzpicture}
