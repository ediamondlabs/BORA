\definecolor{colorNoByzantine}{HTML}{E41A1C}
\definecolor{colorObsNoise}{HTML}{377EB8}
\definecolor{colorObsReplay}{HTML}{4DAF4A}

\pgfplotsset{
    % Shared styles
    ci path/.style={draw=none},
    ci fill/.style={opacity=0.15},
    mean/.style={thick, mark repeat=200, mark size=1.5pt},
    % Per-condition styles
    NoByzantine/.style={color=colorNoByzantine, mark=*},
    ObsNoise/.style={color=colorObsNoise, mark=square*},
    ObsReplay/.style={color=colorObsReplay, mark=triangle*},
}

\begin{tikzpicture}
    \begin{axis}[
        grid=major,
        grid style={dashed, gray!30},
        width=\linewidth,
        xlabel={\small $t$},
        ylabel={\small $\tau$ [\si{\mega\bit\per\second}]},
        legend style={at={(0.5,-0.20)}, anchor=north, legend columns=3},
        tick label style={font=\small},
        label style={font=\small},
    ]

    % ===== No Byzantine =====
    \addplot [name path=NoByzantine upper, ci path, forget plot] table [x=x, y=upNoByzantine, col sep=comma] {steps_combined.csv};
    \addplot [name path=NoByzantine lower, ci path, forget plot] table [x=x, y=lowNoByzantine, col sep=comma] {steps_combined.csv};
    \addplot [NoByzantine, ci fill, forget plot] fill between [of=NoByzantine upper and NoByzantine lower];
    \addplot [NoByzantine, mean] table [x=x, y=meanNoByzantine, col sep=comma] {steps_combined.csv};
    \addlegendentry{No Byzantine}

    % ===== Obs. Noise =====
    \addplot [name path=ObsNoise upper, ci path, forget plot] table [x=x, y=upObsNoise, col sep=comma] {steps_combined.csv};
    \addplot [name path=ObsNoise lower, ci path, forget plot] table [x=x, y=lowObsNoise, col sep=comma] {steps_combined.csv};
    \addplot [ObsNoise, ci fill, forget plot] fill between [of=ObsNoise upper and ObsNoise lower];
    \addplot [ObsNoise, mean] table [x=x, y=meanObsNoise, col sep=comma] {steps_combined.csv};
    \addlegendentry{Obs. Noise}

    % ===== Obs. Replay =====
    \addplot [name path=ObsReplay upper, ci path, forget plot] table [x=x, y=upObsReplay, col sep=comma] {steps_combined.csv};
    \addplot [name path=ObsReplay lower, ci path, forget plot] table [x=x, y=lowObsReplay, col sep=comma] {steps_combined.csv};
    \addplot [ObsReplay, ci fill, forget plot] fill between [of=ObsReplay upper and ObsReplay lower];
    \addplot [ObsReplay, mean] table [x=x, y=meanObsReplay, col sep=comma] {steps_combined.csv};
    \addlegendentry{Obs. Replay}

    \end{axis}
\end{tikzpicture}
