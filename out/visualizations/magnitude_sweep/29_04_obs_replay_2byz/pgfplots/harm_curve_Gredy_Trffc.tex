\definecolor{colorObsNoise}{HTML}{E41A1C}
\definecolor{colorObsReplay}{HTML}{377EB8}

\pgfplotsset{
    ci path/.style={draw=none},
    ci fill/.style={opacity=0.15},
    mean/.style={thick, mark size=1.5pt},
    ObsNoise/.style={color=colorObsNoise, mark=*},
    ObsReplay/.style={color=colorObsReplay, mark=square*},
}

\begin{tikzpicture}
    \begin{axis}[
        grid=major, grid style={dashed, gray!30},
        width=\linewidth,
        xlabel={\small magnitude},
        ylabel={\small $\Delta\tau\,[\%]$},
        legend style={at={(0.5,-0.20)}, anchor=north, legend columns=3},
        tick label style={font=\small}, label style={font=\small},
    ]

    % ===== Obs. Noise =====
    \addplot [name path=ObsNoiseup, ci path, forget plot] table [x=x, y=upObsNoise, col sep=comma] {harm_curve_Gredy_Trffc.csv};
    \addplot [name path=ObsNoiselo, ci path, forget plot] table [x=x, y=lowObsNoise, col sep=comma] {harm_curve_Gredy_Trffc.csv};
    \addplot [ObsNoise, ci fill, forget plot] fill between [of=ObsNoiseup and ObsNoiselo];
    \addplot [ObsNoise, mean] table [x=x, y=meanObsNoise, col sep=comma] {harm_curve_Gredy_Trffc.csv};
    \addlegendentry{Obs. Noise}

    % ===== Obs. Replay =====
    \addplot [name path=ObsReplayup, ci path, forget plot] table [x=x, y=upObsReplay, col sep=comma] {harm_curve_Gredy_Trffc.csv};
    \addplot [name path=ObsReplaylo, ci path, forget plot] table [x=x, y=lowObsReplay, col sep=comma] {harm_curve_Gredy_Trffc.csv};
    \addplot [ObsReplay, ci fill, forget plot] fill between [of=ObsReplayup and ObsReplaylo];
    \addplot [ObsReplay, mean] table [x=x, y=meanObsReplay, col sep=comma] {harm_curve_Gredy_Trffc.csv};
    \addlegendentry{Obs. Replay}

    \end{axis}
\end{tikzpicture}
