    ci fill/.style={opacity=0.15},
    mean/.style={thick, mark size=1.5pt},
    CapacityDeflate/.style={color=colorCapacityDeflate, mark=*},
    DemandDeflate/.style={color=colorDemandDeflate, mark=square*},

\begin{tikzpicture}

        tick label style={font=\small}, label style={font=\small},

    % ===== Capacity Deflate =====
    \addplot [name path=CapacityDeflateup, ci path, forget plot] table [x=x, y=upCapacityDeflate, col sep=comma] {harm_curve_Gredy_Trffc.csv};
